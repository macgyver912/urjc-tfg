Unidades de medida:

\begin{itemize}
	\item Milla náutica (distancias): 1 nm = 1,852 km
	\item Pie (altura / altitud): 1 ft = 0,3048 m = 30,48 cm
	\item Nudo (velocidad): 1 kt = 1,852 km/h
\end{itemize}

Otros:

\begin{itemize}
	\item Altura: distancia vertical sobre el terreno
	\item Altitud: distancia vertical hasta el nivel del mar
	\item Nivel de vuelo: altitud expresada en cientos de pies (FLXXX)
\end{itemize}
